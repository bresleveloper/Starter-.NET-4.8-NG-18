## .Net C# backend:

/// <summary>Divides <paramref name="a"/> by <paramref name="b"/>.</summary>
/// <param name="a">The number to divide.</param>
/// <param name="b">The number to divide by. Must not be zero.</param>
/// <returns>The quotient of the two operands.</returns>
/// <exception cref="DivideByZeroException"><paramref name="b"/> is zero.</exception>
/// <example><code>double result = Calculator.Divide(10, 4); // 2.5</code></example>
public static double Divide(double a, double b)
    => b == 0 ? throw new DivideByZeroException() : a / b;


## angular typescript frontend:

/** Divides `a` by `b`.
* @param a - The number to divide.
* @param b - The number to divide by. Must not be zero.
* @returns The quotient of the two operands.
* @throws {@link RangeError} if `b` is zero.
* @example `calculator.divide(10, 4)` returns `2.5`. */
divide(a: number, b: number): number {
    if (b === 0) { throw new RangeError('Division by zero'); }
    return a / b;
}
